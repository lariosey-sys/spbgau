/*

  Mega2560 + ESP8266(AT) → MQTT контроллер реле (robust)

  Зависимости: WiFiEspAT (by Arduino), PubSubClient (knolleary)

  DIP: режим 5 (ATmega2560 <-> ESP8266), джампер TXD3/RXD3

*/

  

#include <WiFiEspAT.h>

#include <PubSubClient.h>

#include <ctype.h>

#include <string.h>

  

// ---------- НАСТРОЙКИ ----------

const char* WIFI_SSID    = "Greenhouse";

const char* WIFI_PASS    = "77777777";

  

const char* MQTT_HOST    = "192.168.1.112";   // IP брокера

const uint16_t MQTT_PORT = 1883;

const char* MQTT_USER    = "";                // если не нужно — оставить пустым

const char* MQTT_PASS    = "";

  

const char* DEVICE_ID    = "mega-1";          // имя MQTT-клиента

const char* BASE         = "greenhouse";      // корневая тема

// LWT/availability

// greenhouse/mega-1/status -> "online"/"offline" (retain)

char TOPIC_LWT[64];

  

#define ESP_BAUD 115200

#define NUM_RELAYS 15

uint8_t RELAY_PINS[NUM_RELAYS] = {

  22,24,26,28,30,32,34,36,38,40,42,44,46,48,50

};

const bool RELAY_ACTIVE_LOW = true;           // инвертированные входы модулей реле

const bool RELAY_SAFE_OFF_AT_BOOT = true;     // всё выключить при старте (рекомендуется)

const unsigned long SUMMARY_MS = 10000;       // сводка каждые N мс

// --------------------------------

  

WiFiClient wifi;

PubSubClient mqtt(wifi);

bool relayState[NUM_RELAYS] = {0};

unsigned long tSummary = 0;

unsigned long tWiFiRetry = 0;

unsigned long tMqttRetry = 0;

const unsigned long WIFI_RETRY_MS = 2000;

const unsigned long MQTT_RETRY_MS = 2000;

  

// ---- Утилиты ----

inline uint8_t levelFor(bool on) {

  // Уровень на пине: учитываем инверсию

  return ( (RELAY_ACTIVE_LOW ? !on : on) ? HIGH : LOW );

}

  

void applyRelay(uint8_t idx, bool on) {

  if (idx >= NUM_RELAYS) return;

  relayState[idx] = on;

  digitalWrite(RELAY_PINS[idx], levelFor(on));

}

  

void publishRelayState(uint8_t idx) {

  char topic[64];

  snprintf(topic, sizeof(topic), "%s/relay/%u/state", BASE, idx+1);

  mqtt.publish(topic, relayState[idx] ? "ON" : "OFF", /*retain=*/true);

}

  

void publishAllStatesRetained() {

  for (uint8_t i = 0; i < NUM_RELAYS; i++) publishRelayState(i);

}

  

void publishSummary() {

  static char buf[768]; // запас под JSON

  int n = snprintf(buf, sizeof(buf),

                   "{\"ts\":%lu,\"device\":\"%s\",\"relays\":{",

                   millis()/1000UL, DEVICE_ID);

  for (uint8_t i=0; i<NUM_RELAYS; i++) {

    n += snprintf(buf+n, sizeof(buf)-n, "\"%u\":%s%s",

                  i+1, relayState[i] ? "true" : "false",

                  (i<NUM_RELAYS-1)?"," : "");

    if (n >= (int)sizeof(buf)-16) break;

  }

  n += snprintf(buf+n, sizeof(buf)-n, "}}");

  mqtt.publish("greenhouse/relay/summary", buf, n, /*retain=*/false);

}

  

bool equalsNoCase(const char* a, const char* b) {

  while (*a && *b) { if (tolower(*a++) != tolower(*b++)) return false; }

  return *a == *b;

}

  

bool parseBoolCmd(const char* s, bool &val, bool &toggle) {

  toggle = false;

  if (equalsNoCase(s, "ON") || !strcmp(s, "1"))  { val = true;  return true; }

  if (equalsNoCase(s, "OFF")|| !strcmp(s, "0"))  { val = false; return true; }

  if (equalsNoCase(s, "TOGGLE"))                 { toggle = true; return true; }

  return false;

}

  

void setAllCmd(const char* cmd) {

  bool v=false, tog=false;

  if (!parseBoolCmd(cmd, v, tog)) return;

  for (uint8_t i=0; i<NUM_RELAYS; i++) {

    bool on = tog ? !relayState[i] : v;

    applyRelay(i, on);

    publishRelayState(i);

  }

}

  

void replyOne(uint8_t idx) {

  publishRelayState(idx);

}

  

// greenhouse/relay/<N|all>/(set|get)

void onMqtt(char* topic, byte* payload, unsigned int len) {

  // нормализуем payload в C-строку

  char cmd[16];

  len = (len >= sizeof(cmd)) ? sizeof(cmd)-1 : len;

  memcpy(cmd, payload, len); cmd[len]=0;

  

  // разберём «хвост» темы

  // ожидается: BASE/relay/<token>/<action>

  // пример:   greenhouse/relay/1/set

  //           greenhouse/relay/all/get

  const char* p = topic;

  

  // Пропускаем BASE/relay/

  char expect[32];

  snprintf(expect, sizeof(expect), "%s/relay/", BASE);

  size_t prefixLen = strlen(expect);

  if (strncmp(p, expect, prefixLen) != 0) return;

  p += prefixLen;

  

  // Выделяем token (до следующего '/')

  char token[12] = {0};

  const char* slash = strchr(p, '/');

  if (!slash) return;

  size_t tokLen = (size_t)(slash - p);

  if (tokLen >= sizeof(token)) tokLen = sizeof(token)-1;

  memcpy(token, p, tokLen); token[tokLen]=0;

  

  // action

  const char* action = slash + 1; // "set" или "get"

  if (equalsNoCase(action, "set")) {

    if (equalsNoCase(token, "all")) {

      setAllCmd(cmd);

      return;

    }

    int n = atoi(token);

    if (n >= 1 && n <= (int)NUM_RELAYS) {

      bool v=false, tog=false;

      if (!parseBoolCmd(cmd, v, tog)) return;

      bool newVal = tog ? !relayState[n-1] : v;

      applyRelay(n-1, newVal);

      publishRelayState(n-1);

    }

  } else if (equalsNoCase(action, "get")) {

    if (equalsNoCase(token, "all")) {

      publishAllStatesRetained();

      publishSummary();

      return;

    }

    int n = atoi(token);

    if (n >= 1 && n <= (int)NUM_RELAYS) {

      replyOne(n-1);

    }

  }

}

  

void wifiInitOnce() {

  // иногда модем «просыпается» не сразу: дадим немного времени

  for (int i=0; i<10 && WiFi.status()==WL_NO_SHIELD; i++) delay(200);

  // WiFiEspAT не имеет setTimeout(); таймаут реализован в ensureWiFi()

}

  
  

void ensureWiFi() {

  if (WiFi.status() == WL_CONNECTED) return;

  unsigned long now = millis();

  if (now - tWiFiRetry < WIFI_RETRY_MS) return;

  tWiFiRetry = now;

  

  WiFi.begin(WIFI_SSID, WIFI_PASS);

  unsigned long t0 = now;

  while (WiFi.status() != WL_CONNECTED && millis() - t0 < 20000) {

    delay(250);

  }

}

  

void ensureMQTT() {

  if (mqtt.connected()) return;

  unsigned long now = millis();

  if (now - tMqttRetry < MQTT_RETRY_MS) return;

  tMqttRetry = now;

  

  mqtt.setServer(MQTT_HOST, MQTT_PORT);

  mqtt.setCallback(onMqtt);

  mqtt.setBufferSize(768);

  mqtt.setKeepAlive(30);

  mqtt.setSocketTimeout(20);

  

  // Тема LWT

  snprintf(TOPIC_LWT, sizeof(TOPIC_LWT), "%s/%s/status", BASE, DEVICE_ID);

  

  // LWT: offline (retain). При connect сразу опубликуем online.

  bool ok;

  if (MQTT_USER && MQTT_USER[0]) {

    ok = mqtt.connect(DEVICE_ID, MQTT_USER, MQTT_PASS, TOPIC_LWT, 1, true, "offline");

  } else {

    ok = mqtt.connect(DEVICE_ID, TOPIC_LWT, 1, true, "offline");

  }

  

  if (ok) {

    mqtt.publish(TOPIC_LWT, "online", true);

    mqtt.subscribe("greenhouse/relay/+/set");

    mqtt.subscribe("greenhouse/relay/+/get");

    // раздать текущие состояния (retain) + сводку

    publishAllStatesRetained();

    publishSummary();

  }

}

  

void setup() {

  Serial.begin(115200);

  Serial3.begin(ESP_BAUD);

  WiFi.init(Serial3);

  wifiInitOnce();

  

  // Инициализация пинов реле

  for (uint8_t i=0; i<NUM_RELAYS; i++) {

    pinMode(RELAY_PINS[i], OUTPUT);

    bool initial = RELAY_SAFE_OFF_AT_BOOT ? false : relayState[i];

    applyRelay(i, initial);

  }

  

  ensureWiFi();

  ensureMQTT();

}

  

void loop() {

  if (WiFi.status() != WL_CONNECTED) ensureWiFi();

  if (!mqtt.connected()) ensureMQTT();

  mqtt.loop();

  

  if (millis() - tSummary >= SUMMARY_MS) {

    tSummary = millis();

    publishSummary();

  }

}
